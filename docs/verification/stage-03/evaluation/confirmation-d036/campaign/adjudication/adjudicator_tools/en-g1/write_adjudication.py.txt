def locations(obj, path=()):
    if isinstance(obj, dict):
        if set(obj.keys()) >= {"cp", "quote"} and isinstance(obj.get("cp"), int):
            yield path, obj
        for k, v in obj.items():
            yield from locations(v, path + (k,))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            yield from locations(v, path + (i,))

def check_quotes(text, obj, label):
    bad = []
    for p, loc in locations(obj):
        if text[loc["cp"]:loc["cp"] + len(loc["quote"])] != loc["quote"]:
            bad.append((p, loc))
    if bad:
        raise SystemExit(f"{label}: quote mismatch {bad}")
