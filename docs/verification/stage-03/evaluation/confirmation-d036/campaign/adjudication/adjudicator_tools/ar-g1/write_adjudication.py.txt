def main():
    for article, decisions in ARTICLES.items():
        dec_path = os.path.join(BASE, "out", "decisions", article + ".json")
        adj_path = os.path.join(BASE, "out", "adjudicated", article + ".json")
        src = os.path.join(BASE, "items", article, "locked.json")

        decision_doc = {
            "article": article,
            "adjudicator": ADJ,
            "started_at": STARTED,
            "finished_at": FINISHED,
            "decisions": decisions,
        }
        with open(dec_path, "w", encoding="utf-8") as f:
            json.dump(decision_doc, f, ensure_ascii=False, indent=2)
            f.write("\n")

        with open(src, encoding="utf-8") as f:
            locked = json.load(f)
        adjusted = copy.deepcopy(locked)
        changed = []
        for d in decisions:
            if d["decision"] != "correct":
                continue
            corr = d["corrected_label"]
            if corr["scheme"] == "doi" and "identifiers" in adjusted:
                matches = [
                    i for i, x in enumerate(adjusted["identifiers"])
                    if x["at"] == corr["at"]
                ]
                assert len(matches) == 1, (article, d["item"], matches)
                idx = matches[0]
                old = adjusted["identifiers"][idx]
                changed.append({
                    "item": d["item"],
                    "kind": d["kind"],
                    "identifier_index": idx,
                    "location": corr["at"],
                    "field": "key",
                    "from": old["key"],
                    "to": corr["key"],
                })
                old["key"] = corr["key"]
        adjusted["adjudication"] = {
            "adjudicator": ADJ,
            "decisions_file": dec_path,
            "changed_items": changed,
        }
        with open(adj_path, "w", encoding="utf-8") as f:
            json.dump(adjusted, f, ensure_ascii=False, indent=2)
            f.write("\n")
        print(article, "decisions", len(decisions), "changed", len(changed))


if __name__ == "__main__":
    main()
