def load(rev):
    p=json.load(open(f'{B}/packets/en-{rev}/packet.json'))
    t=open(f'{B}/packets/en-{rev}/source.wikitext',encoding='utf-8').read()
    return p,t
def L(t,cp,n=40):
    q=t[cp:cp+n]
    assert t[cp:cp+len(q)]==q
    return {"cp":cp,"quote":q}
def find(t,needle,start=0):
    i=t.find(needle,start); assert i>=0,needle; return i
def sups(rev):
    d=html.parse(f'{B}/packets/en-{rev}/render.html').getroot()
    return [s.get('id') for s in d.iter('sup') if 'mw:Extension/ref' in (s.get('typeof') or '')]

def D(i,t,cp,kind,name,group,templates,unstr,note_id,notes="",in_window=True,n=30,marker="none"):
    return {"id":i,"at":L(t,cp,n),"in_window":in_window,"kind":kind,"name":name,"group":group,
      "templates":[{"at":L(t,c,n)} for c in templates],"unstructured":unstr,"render_note_id":note_id,
      "marker":marker,"marker_text":None,"notes":notes}
def U(i,t,cp,bind,rid,notes="",n=30):
    return {"id":i,"at":L(t,cp,n),"binds_to":(L(t,bind,n) if isinstance(bind,int) else bind),"render_ref_id":rid,"notes":notes}
def I(t,cp,txt,scheme,role,key,notes=""):
    assert t[cp:cp+len(txt)]==txt
    return {"at":{"cp":cp,"quote":txt},"scheme":scheme,"role":role,"key":key,"notes":notes}
