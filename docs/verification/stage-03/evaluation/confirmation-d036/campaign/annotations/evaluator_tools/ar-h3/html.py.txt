import sys
from lxml import html
d=sys.argv[1]
r=html.parse(f'packets/{d}/render.html').getroot()
for s in r.xpath('//sup[contains(@typeof,"mw:Extension/ref")]'):
    print('SUP',s.get('id'),(s.get('data-mw') or '')[:150].replace('\n',' '))
for l in r.xpath('//li[starts-with(@id,"cite_note")]'):
    print('LI',l.get('id'),' '.join(l.text_content().split())[:160])
