import json,hashlib
ROOT='/tmp/claude-0/-home-user-citerift/a52d5c86-81c2-5300-a98c-c54b1eaf0da5/scratchpad/eval-d036/ar-h1/'
class Art:
    def __init__(s,d):
        s.d=d; b=ROOT+'packets/'+d+'/'
        s.t=open(b+'source.wikitext',encoding='utf-8').read()
        s.p=json.load(open(b+'packet.json'))
        s.defs=[];s.uses=[];s.ids=[];s.un=[]
    def L(s,needle,nth=0,after=0,n=60):
        i=after-1
        for _ in range(nth+1):
            i=s.t.find(needle,i+1)
            if i<0: raise Exception('nf '+needle)
        return {'cp':i,'quote':s.t[i:i+max(len(needle),min(n,len(s.t)-i))][:max(len(needle),n)]}
    def Lq(s,cp,n=60):
        return {'cp':cp,'quote':s.t[cp:cp+n]}
    def inw(s,cp): return s.p['window']['cp_start']<=cp<s.p['window']['cp_end']
    def D(s,id,at,kind,name,group,templates,unstructured,rid,marker='none',mt=None,notes='',in_window=None):
        s.defs.append(dict(id=id,at=at,in_window=s.inw(at['cp']) if in_window is None else in_window,kind=kind,name=name,group=group,
          templates=[{'at':x} for x in templates],unstructured=unstructured,render_note_id=rid,marker=marker,marker_text=mt,notes=notes))
    def U(s,id,at,bind,rid,notes=''):
        s.uses.append(dict(id=id,at=at,binds_to=bind,render_ref_id=rid,notes=notes))
    def I(s,at,scheme,role,key,notes=''):
        s.ids.append(dict(at=at,scheme=scheme,role=role,key=key,notes=notes))
    def write(s,files,start,tools,unresolved,un=None):
        import datetime
        for x in s.defs+s.uses+s.ids:
            for loc in [x['at']]+([x['binds_to']] if 'binds_to' in x and isinstance(x['binds_to'],dict) else [])+[y['at'] for y in x.get('templates',[])]:
                assert s.t[loc['cp']:loc['cp']+len(loc['quote'])]==loc['quote'],(loc)
        w=s.p['window']
        o={'format':'citerift-stage03-annotation-v1',
         'article':{'wiki':s.p['wiki'],'revision_id':s.p['revision_id'],'source_sha256':s.p['source_sha256'],'html_sha256':s.p['html_sha256'],'window':[w['cp_start'],w['cp_end']]},
         'provenance':{'agent':'d036-eval-ar-h1','started_at':start,'finished_at':datetime.datetime.utcnow().strftime('%Y-%m-%dT%H:%M:%SZ'),
           'files_opened':[{'path':'ANNOTATION_GUIDE_v1.1.md','sha256':'2b40bb323c2254f8c12e40f9de39e759abae6b89aba8ffcc916b8abcb648cf2e'},
             {'path':'packets/%s/source.wikitext'%s.d,'sha256':s.p['source_sha256']},{'path':'packets/%s/render.html'%s.d,'sha256':s.p['html_sha256']},{'path':'packets/%s/packet.json'%s.d,'sha256':hashlib.sha256(open(ROOT+'packets/'+s.d+'/packet.json','rb').read()).hexdigest()}],
           'tools':tools,'unresolved':unresolved},
         'definitions':s.defs,'uses':s.uses,'identifiers':s.ids,'unresolved_notes':s.un}
        json.dump(o,open(ROOT+'out/%s-%d.json'%(s.p['wiki'],s.p['revision_id']),'w',encoding='utf-8'),ensure_ascii=False,indent=1)
        print(s.d,len(s.defs),len(s.uses),len(s.ids))
