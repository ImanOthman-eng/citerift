import json,re,sys
from hashlib import sha256
class A:
    def __init__(s,d):
        s.d=d
        s.t=open(f'packets/{d}/source.wikitext',encoding='utf-8').read()
        s.p=json.load(open(f'packets/{d}/packet.json'))
        s.defs=[];s.uses=[];s.ids=[];s.un=[]
    def L(s,q,n=0,after=0):
        i=after-1
        for _ in range(n+1):
            i=s.t.find(q,i+1)
            assert i>=0,(q,n)
        assert s.t[i:i+len(q)]==q
        return {'cp':i,'quote':q}
    def inw(s,l): return s.p['window']['cp_start']<=l['cp']<s.p['window']['cp_end']
    def D(s,id,at,kind,name=None,group='',templates=(),unstructured=False,rid=None,marker='none',mt=None,notes='',inw=None):
        s.defs.append(dict(id=id,at=at,in_window=s.inw(at) if inw is None else inw,kind=kind,name=name,group=group,templates=[{'at':t} for t in templates],unstructured=unstructured,render_note_id=rid,marker=marker,marker_text=mt,notes=notes))
    def U(s,id,at,bind,rid=None,notes=''):
        s.uses.append(dict(id=id,at=at,binds_to=bind,render_ref_id=rid,notes=notes))
    def I(s,at,scheme,role,key,notes=''):
        s.ids.append(dict(at=at,scheme=scheme,role=role,key=key,notes=notes))
    def save(s,unres='',started='',files=()):
        p=s.p
        out={'format':'citerift-stage03-annotation-v1',
         'article':{'wiki':p['wiki'],'revision_id':p['revision_id'],'source_sha256':p['source_sha256'],'html_sha256':p['html_sha256'],'window':[p['window']['cp_start'],p['window']['cp_end']]},
         'provenance':{'agent':'d036-eval-ar-h2','started_at':'2026-11-04T10:24:00Z','finished_at':__import__('datetime').datetime.utcnow().strftime('%Y-%m-%dT%H:%M:%SZ'),
          'files_opened':[{'path':'ANNOTATION_GUIDE_v1.1.md','sha256':'2b40bb323c2254f8c12e40f9de39e759abae6b89aba8ffcc916b8abcb648cf2e'},
            {'path':f'packets/{s.d}/packet.json','sha256':sha256(open(f'packets/{s.d}/packet.json','rb').read()).hexdigest()},
            {'path':f'packets/{s.d}/source.wikitext','sha256':p['source_sha256']},{'path':f'packets/{s.d}/render.html','sha256':p['html_sha256']}],
          'tools':'python3 -I stdlib + lxml: regex scan of source, lxml listing of sup/li elements; locations verified by slicing',
          'unresolved':unres},
         'definitions':s.defs,'uses':s.uses,'identifiers':s.ids,'unresolved_notes':s.un}
        # verify
        def chk(l):
            if isinstance(l,dict) and 'cp' in l: assert s.t[l['cp']:l['cp']+len(l['quote'])]==l['quote'],l
        for x in s.defs+s.uses+s.ids:
            chk(x['at']); chk(x.get('binds_to'))
            for t in x.get('templates',[]): chk(t['at'])
        assert hashlib_ok(p,s)
        json.dump(out,open(f'out/ar-{p["revision_id"]}.json','w'),ensure_ascii=False,indent=1)
        print(s.d,'defs',len(s.defs),'uses',len(s.uses),'ids',len(s.ids))
def hashlib_ok(p,s): return sha256(s.t.encode()).hexdigest()==p['source_sha256']
