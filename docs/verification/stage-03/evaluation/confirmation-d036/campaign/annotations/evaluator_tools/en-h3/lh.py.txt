import sys
from lxml import html
d=sys.argv[1]
doc=html.parse(f'packets/{d}/render.html').getroot()
for s in doc.xpath('//sup[contains(@typeof,"mw:Extension/ref")]'):
    print('SUP',s.get('id'),(s.get('data-mw') or '')[:230].replace('\n',' '))
for li in doc.xpath('//li[starts-with(@id,"cite_note")]'):
    print('LI',li.get('id'),' '.join(li.text_content().split())[:260])
