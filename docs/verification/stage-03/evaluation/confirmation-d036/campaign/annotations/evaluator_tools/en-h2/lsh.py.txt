import sys,re
from lxml import html
d=html.parse(sys.argv[1]).getroot()
for s in d.iter('sup'):
    if 'mw:Extension/ref' in (s.get('typeof') or ''):
        print('SUP',s.get('id'),(s.get('data-mw') or '')[:150].replace('\n',' '))
for l in d.iter('li'):
    if (l.get('id') or '').startswith('cite_note'):
        print('LI',l.get('id'),' '.join(l.text_content().split())[:200])
