import sys,re,json
d=sys.argv[1]
t=open(f'packets/{d}/source.wikitext',encoding='utf-8').read()
p=json.load(open(f'packets/{d}/packet.json'))
s,e=p['window']['cp_start'],p['window']['cp_end']
print(len(t),s,e)
pat=re.compile(r'<ref\b[^>]*>|</ref>|\{\{\s*(?:cite|استشهاد|sfn|harv|efn|refn|ملا|doi|pmid|دوي|r\||reflist|مراجع|ثبت)[^|}]*|10\.\d{4,9}/|PMID|pmid|doi|دوي|<!--|<nowiki|\{\{\s*(?:مصدر|retract)',re.I)
for m in pat.finditer(t):
    if s<=m.start()<e: print(m.start(),repr(t[m.start():m.start()+70]))
