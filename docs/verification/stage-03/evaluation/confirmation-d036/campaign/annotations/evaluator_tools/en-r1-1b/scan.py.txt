import re,sys
d=sys.argv[1]
t=open(f'packets/{d}/source.wikitext',encoding='utf-8').read()
pat=re.compile(r'<ref\b|</ref>|\{\{\s*(?:cite|citation|sfn|sfnp|efn|refn|r\||harv|doi|pmid|retract|expression|reflist|notelist)|10\.\d{4,9}/|PMID|pubmed|doi',re.I)
for m in pat.finditer(t):
    print(m.start(),repr(t[m.start():m.start()+70]))
