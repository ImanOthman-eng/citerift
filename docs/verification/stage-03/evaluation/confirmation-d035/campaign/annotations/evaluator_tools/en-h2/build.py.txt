def load(rev):

def loc(t,cp,n=30):
    q=t[cp:cp+n]; assert t[cp:cp+len(q)]==q; return {'cp':cp,'quote':q}
def ref_end(t,cp):
    return t.index('</ref>',cp)
def tpl_locs(t,s,e):
    return [{'at':loc(t,m.start(),20)} for m in re.finditer(r'\{\{\s*(?:cite|Cite)\b',t[s:e]) for m in [type('M',(),{'start':lambda self,o=s+m.start():o})()]]
def ids_in(t,s,e,role='primary_citation_field'):
    out=[]
    for m in re.finditer(r'\|\s*(doi|pmid)\s*=\s*([^\n|}]*?)\s*(?=[\n|}])',t[s:e]):
        sch=m.group(1); val=m.group(2); cp=s+m.start(2)
        assert t[cp:cp+len(val)]==val
        if sch=='doi': key=val.lower() if re.fullmatch(r'10\.\d{4,9}/\S+',val) else None
        else: key=val if re.fullmatch(r'[1-9]\d{0,8}',val) else None
        out.append({'at':{'cp':cp,'quote':val},'scheme':sch,'role':role,'key':key,'notes':''})
    return out
