def sha(p): return hashlib.sha256(open(p,'rb').read()).hexdigest()

class Art:
    def __init__(s,rev):
        s.rev=rev; d=f'{BASE}/packets/ar-{rev}'; s.dirp=d
        s.pk=json.load(open(d+'/packet.json'))
        s.t=open(d+'/source.wikitext',encoding='utf-8').read()
        assert sha(d+'/source.wikitext')==s.pk['source_sha256']
        s.has_html=s.pk['html_sha256'] is not None
        s.sups=[];s.lis=[]
        if s.has_html:
            assert sha(d+'/render.html')==s.pk['html_sha256']
            r=html.parse(d+'/render.html').getroot()
            for e in r.iter('sup'):
                if 'mw:Extension/ref' in (e.get('typeof') or ''):
                    dm=json.loads(e.get('data-mw')); s.sups.append((e.get('id'),dm))
            s.lis=[e.get('id') for e in r.iter('li') if (e.get('id') or '').startswith('cite_note')]
        s.defs=[];s.uses=[];s.ids=[];s.unres=[]
    def pos(s,sub,n=0,start=0):
        i=start-1
        for _ in range(n+1):
            i=s.t.find(sub,i+1); assert i>=0,(sub,n)
        return i
    def loc(s,cp,ln=40):
        q=s.t[cp:cp+ln]; assert s.t[cp:cp+len(q)]==q; return {'cp':cp,'quote':q}
    def note_id(s,si):
        if si is None: return None
        dm=s.sups[si][1]; b=dm.get('body',{})
        if 'id' in b: return b['id'].replace('mw-reference-text-','')
        return None
    def d(s,cp,kind,name,group,tpls,unstr,si=None,li=None,marker=None,mtext=None,notes='',inw=True,ln=40):
        did=f'd{len(s.defs)+1}'
        nid=li if li else s.note_id(si)
        if marker is None: marker='none' if s.has_html else 'cannot_tell'
        s.defs.append({'id':did,'at':s.loc(cp,ln),'in_window':inw,'kind':kind,'name':name,'group':group,
          'templates':[{'at':s.loc(c,24)} for c in tpls],'unstructured':unstr,'render_note_id':nid,
          'marker':marker,'marker_text':mtext,'notes':notes})
    def u(s,cp,bind,si=None,notes='',ln=40):
        rid=s.sups[si][0] if (si is not None and s.has_html) else None
        s.uses.append({'id':f'u{len(s.uses)+1}','at':s.loc(cp,ln),'binds_to':s.loc(bind,ln) if bind!='missing' else 'missing',
          'render_ref_id':rid,'notes':notes})
    def i(s,cp,ln,scheme,role,key,notes=''):
        raw=s.t[cp:cp+ln]
        s.ids.append({'at':{'cp':cp,'quote':raw},'scheme':scheme,'role':role,'key':key,'notes':notes})
    def write(s,tools,unresolved):
        w=s.pk['window']; w=[w['cp_start'],w['cp_end']]
        for L in [x['at'] for x in s.defs+s.uses+s.ids]:
            assert s.t[L['cp']:L['cp']+len(L['quote'])]==L['quote']
        files=[{'path':'ANNOTATION_GUIDE_v1.1.md','sha256':GUIDE},
               {'path':f'packets/ar-{s.rev}/packet.json','sha256':sha(s.dirp+'/packet.json')},
               {'path':f'packets/ar-{s.rev}/source.wikitext','sha256':s.pk['source_sha256']}]
        if s.has_html: files.append({'path':f'packets/ar-{s.rev}/render.html','sha256':s.pk['html_sha256']})
        out={'format':'citerift-stage03-annotation-v1',
         'article':{'wiki':'ar','revision_id':s.rev,'source_sha256':s.pk['source_sha256'],'html_sha256':s.pk['html_sha256'],'window':w},
         'provenance':{'agent':'d035-eval-ar-h1','started_at':T0,'finished_at':'2026-11-03T10:00:00Z','files_opened':files,
           'tools':'python3 -I standard library + lxml: str.find/regex to locate tokens, lxml to list sup/li elements; every location verified by slicing; all decisions by hand',
           'unresolved':unresolved},
         'definitions':s.defs,'uses':s.uses,'identifiers':s.ids,'unresolved_notes':[unresolved] if unresolved else []}
        json.dump(out,open(f'{BASE}/out/ar-{s.rev}.json','w'),ensure_ascii=False,indent=1)
        print(s.rev,'defs',len(s.defs),'uses',len(s.uses),'ids',len(s.ids),'window',w)
