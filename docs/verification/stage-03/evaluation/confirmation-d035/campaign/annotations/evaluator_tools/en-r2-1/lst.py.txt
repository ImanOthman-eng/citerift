import sys,lxml.html
d=lxml.html.parse(sys.argv[1]).getroot()
for s in d.xpath('//sup[contains(@typeof,"mw:Extension/ref")]'):
    print('SUP',s.get('id'),(s.get('data-mw') or '')[:150].replace('\n',' '))
for l in d.xpath('//li[starts-with(@id,"cite_note")]'):
    print('LI',l.get('id'),' '.join(l.text_content().split())[:200])
