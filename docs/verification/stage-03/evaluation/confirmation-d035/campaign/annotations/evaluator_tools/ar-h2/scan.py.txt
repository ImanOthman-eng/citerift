import sys,re
d=sys.argv[1]
t=open(f'../packets/{d}/source.wikitext',encoding='utf-8').read()
pat=re.compile(r'<ref|</ref|\{\{\s*(?:cite|citation|استشهاد|sfn|efn|refn|ملا|r\b|harv|doi|دوي|pmid)|10\.\d{4}|PMID|pmid|doi|دوي|<!--|<nowiki|\{\{\s*(?:Reflist|مراجع|ثبت)|^[*#:;]',re.I|re.M)
for m in pat.finditer(t):
    print(m.start(),repr(t[m.start():m.start()+70]))
