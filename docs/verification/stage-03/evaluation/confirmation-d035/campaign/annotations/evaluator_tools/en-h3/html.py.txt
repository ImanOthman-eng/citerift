import sys,json
from lxml import html
d=sys.argv[1]
doc=html.parse(f'packets/{d}/render.html')
for s in doc.xpath('//sup[contains(@typeof,"mw:Extension/ref")]'):
    dm=s.get('data-mw') or ''
    print('SUP',s.get('id'),dm[:int(sys.argv[2]) if len(sys.argv)>2 else 150].replace('\n',' '))
for li in doc.xpath('//li[starts-with(@id,"cite_note")]'):
    print('LI',li.get('id'),' '.join(li.text_content().split())[:int(sys.argv[3]) if len(sys.argv)>3 else 100])
