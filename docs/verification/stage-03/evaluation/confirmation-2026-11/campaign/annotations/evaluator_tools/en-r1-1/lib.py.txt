import json,hashlib,datetime
class A:
    def __init__(s,d):
        s.d=d; s.t=open(f'packets/{d}/source.wikitext',encoding='utf-8').read()
        s.p=json.load(open(f'packets/{d}/packet.json')); s.defs=[];s.uses=[];s.ids=[];s.un=[]
        s.start=datetime.datetime.utcnow().strftime('%Y-%m-%dT%H:%M:%SZ')
    def at(s,q,frm=0,n=0,cp=None):
        if cp is None:
            i=frm-1
            for _ in range(n+1):
                i=s.t.find(q,i+1)
                assert i>=0,(q,frm,n)
        else: i=cp
        assert s.t[i:i+len(q)]==q
        return {"cp":i,"quote":s.t[i:i+60] if len(q)<8 else q[:60]} if True else None
    def loc(s,q,frm=0,n=0):
        i=s.t.find(q,frm)
        for _ in range(n): i=s.t.find(q,i+1)
        assert i>=0,q
        qq=s.t[i:i+max(len(q),min(40,len(q)+20))] if len(q)<8 else q
        qq=s.t[i:i+max(len(q),12)]
        assert s.t[i:i+len(qq)]==qq
        return {"cp":i,"quote":qq}
    def inw(s,cp): return s.p['window']['cp_start']<=cp<s.p['window']['cp_end']
    def define(s,id,loc,kind,name,group,tmpl,unstr,note_id,marker='none',mt=None,notes=''):
        s.defs.append(dict(id=id,at=loc,in_window=s.inw(loc['cp']),kind=kind,name=name,group=group,templates=[{"at":x} for x in tmpl],unstructured=unstr,render_note_id=note_id,marker=marker,marker_text=mt,notes=notes))
    def use(s,id,loc,bind,rid,notes=''):
        s.uses.append(dict(id=id,at=loc,binds_to=bind,render_ref_id=rid,notes=notes))
    def ident(s,loc,scheme,role,key,notes=''):
        s.ids.append(dict(at=loc,scheme=scheme,role=role,key=key,notes=notes))
    def write(s,unresolved=''):
        w=s.p['window']
        for L in [x['at'] for x in s.defs+s.uses+s.ids]+[x['at'] for d in s.defs for x in d['templates']]+[u['binds_to'] for u in s.uses if isinstance(u['binds_to'],dict)]:
            assert s.t[L['cp']:L['cp']+len(L['quote'])]==L['quote'],L
        out={"format":"citerift-stage03-annotation-v1",
         "article":{"wiki":s.p['wiki'],"revision_id":s.p['revision_id'],"source_sha256":s.p['source_sha256'],"html_sha256":s.p['html_sha256'],"window":[w['cp_start'],w['cp_end']]},
         "provenance":{"agent":"cf11-eval-en-r1-1","started_at":s.start,"finished_at":datetime.datetime.utcnow().strftime('%Y-%m-%dT%H:%M:%SZ'),
           "files_opened":[{"path":"ANNOTATION_GUIDE_v1.1.md","sha256":"2b40bb323c2254f8c12e40f9de39e759abae6b89aba8ffcc916b8abcb648cf2e"},
             {"path":f"packets/{s.d}/source.wikitext","sha256":s.p['source_sha256']},{"path":f"packets/{s.d}/render.html","sha256":s.p['html_sha256']},{"path":f"packets/{s.d}/packet.json","sha256":hashlib.sha256(open(f'packets/{s.d}/packet.json','rb').read()).hexdigest()}],
           "tools":"python3 stdlib + lxml helper snippets (regex search for refs/templates/identifiers, HTML sup/li listing, quote verification); labels decided by hand","unresolved":unresolved},
         "definitions":s.defs,"uses":s.uses,"identifiers":s.ids,"unresolved_notes":s.un}
        json.dump(out,open(f"out/{s.p['wiki']}-{s.p['revision_id']}.json",'w'),ensure_ascii=False,indent=1)
        print(s.d,len(s.defs),len(s.uses),len(s.ids))
