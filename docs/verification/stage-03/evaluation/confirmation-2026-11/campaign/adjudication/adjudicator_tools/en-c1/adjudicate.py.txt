def erratum_text(article):
    raw = open(f'{ITEMS}/{article}/render.html', 'rb').read()
    doc = html.fromstring(raw)
    for li in doc.iter('li'):
        if li.get('id', '').startswith('cite_note-GaiaEDR3-'):
            txt = li.text_content()
            s = txt.find('(Erratum:')
            e = txt.find(')', s) + 1
            return txt[s:e]
    raise SystemExit('no GaiaEDR3 footnote in ' + article)

GAIA_ERRATUM_REASON = (
    "The rendered footnote (cite_note-GaiaEDR3) carries '(Erratum: doi:10.1051/0004-6361/202039657e)', "
    "generated by the Cite Gaia EDR3 template for the cited Gaia EDR3 paper. It states no retraction and no "
    "expression of concern, so 'warning' does not apply. The locked 'none' asserts that no retraction or correction "
    "vocabulary is present in the rendered footnote; that is not so. Section 3.6 names 'erratum' as ambiguous "
    "vocabulary and defines 'none' as absence of these, so 'ambiguous' is the closest reading of the guide. "
    "Judgment call: the erratum relation is to the cited work itself and could be read as settled, which would give "
    "'none'. Flagged for the owner's review."
)

def use_reason(name, cp):
    return ("Source has <ref name=%s>...</ref> with content at cp %d inside the <references> list, "
            "or inside {{Reflist|refs=}}. The guide counts each <ref> with content as a use (3.2, and the "
            "definitions table: one <ref> with content gives 1 definition and 1 use). The locked use binds to itself "
            "and its render_ref_id null is right, because a list-defined note emits no inline marker. The engine "
            "missed it. Confirm." % (name, cp))

def build(article, decisions):
    disputed = json.load(open(f'{ITEMS}/{article}/disputed_items.json', encoding='utf-8'))
    ids = [d['item'] for d in disputed]
    assert sorted(decisions) == sorted(ids), (article, sorted(decisions), ids)
    return disputed
