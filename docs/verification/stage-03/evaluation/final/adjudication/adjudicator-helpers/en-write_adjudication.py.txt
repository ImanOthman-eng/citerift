def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def load(a):
    d = json.load(open(f"{BASE}/items/en/{a}/disputed_items.json", encoding="utf-8"))
    L = json.load(open(d["locked_annotation"], encoding="utf-8"))
    t = open(d["packet_dir"] + "/source.wikitext", encoding="utf-8").read()
    return d, L, t

def check_locs(obj, t, path="$"):
    if isinstance(obj, dict):
        if set(obj) >= {"cp", "quote"} and isinstance(obj.get("cp"), int):
            assert t[obj["cp"]:obj["cp"] + len(obj["quote"])] == obj["quote"], (path, obj)
        for k, v in obj.items():
            check_locs(v, t, f"{path}.{k}")
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            check_locs(v, t, f"{path}[{i}]")

SPRINGER_REASON = (
    "The DOI occurrence itself is in scope (3.5: every occurrence, also in URLs) and its role is "
    "identifier_link (a publisher URL in url= pointing at the identifier), but its raw text is wrong. "
    "The raw-text table's URL row lists the URL forms whose whole URL is quoted: doi.org, dx.doi.org, "
    "the two PubMed forms and 'a publisher URL whose path contains /doi/10.…'. This link.springer.com URL "
    "has the DOI directly after the host (/10.1007/…) and no /doi/ segment, so it is not one of the listed "
    "URL forms and falls under 'anything else: the identifier text'. Ambiguity: the URL row could be read as "
    "examples rather than a closed list; the closer reading of its wording (a specific path condition, "
    "repeated in the key rule 'the DOI from the /doi/ path segment') is that it is closed. The item's start "
    "therefore moves to the first character of the DOI; key unchanged."
)

def springer(a, item_no, url_prefix):
    d, L, t = load(a)
    it = [i for i in d["items"] if i["item"] == item_no][0]
    old = it["annotated"]
    idx = [k for k, x in enumerate(L["identifiers"]) if x["at"] == old["at"]]
    assert len(idx) == 1
    url = old["at"]["quote"]
    assert url.startswith(url_prefix)
    doi = url[len(url_prefix):]
    cp = old["at"]["cp"] + len(url_prefix)
    new = copy.deepcopy(L["identifiers"][idx[0]])
    new["at"] = {"cp": cp, "quote": doi}
    assert new["key"] == doi.lower() if doi.isascii() else True
    new["notes"] = ("Adjudicated: DOI written directly in the path of a link.springer.com publisher URL in url= "
                    "(no /doi/ segment). Not one of the guide's listed URL forms, so the raw text is the identifier "
                    "text (guide 3.5 'anything else'); role identifier_link because it sits in a publisher URL "
                    "inside the citation template pointing at the identifier.")
    adj = copy.deepcopy(L)
    adj["identifiers"][idx[0]] = new
    decisions = [{"item": item_no, "kind": it["kind"], "decision": "correct",
                  "reason": SPRINGER_REASON, "corrected_label": new}]
    return d, L, t, adj, decisions, [item_no]

def confirm_all(a, reasons):
    d, L, t = load(a)
    decisions = []
    for it in d["items"]:
        decisions.append({"item": it["item"], "kind": it["kind"], "decision": "confirm",
                          "reason": reasons[it["item"]], "corrected_label": None})
    assert len(decisions) == len(reasons)
    return d, L, t, copy.deepcopy(L), decisions, []
