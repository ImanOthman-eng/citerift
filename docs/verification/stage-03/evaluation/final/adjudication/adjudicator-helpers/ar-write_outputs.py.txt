def check(doc,text):
    bad=[]
    def walk(o):
        if isinstance(o,dict):
            if 'cp' in o and 'quote' in o and isinstance(o['cp'],int):
                if text[o['cp']:o['cp']+len(o['quote'])]!=o['quote']: bad.append(o)
            for v in o.values(): walk(v)
        elif isinstance(o,list):
            for v in o: walk(v)
    walk(doc); return bad
