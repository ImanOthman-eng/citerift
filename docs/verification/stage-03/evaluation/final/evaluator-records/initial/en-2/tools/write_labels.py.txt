def loc(cp,prefix=None,raw=None):
    if prefix is not None: assert text.startswith(prefix,cp),(cp,prefix,text[cp:cp+60])
    quote=raw if raw is not None else text[cp:cp+60]
    assert text[cp:cp+len(quote)]==quote
    return {'cp':cp,'quote':quote}

def definition(cp,kind='note',name=None,templates=(),note_id=None,unstructured=False,notes=''):
    at=loc(cp,'*' if kind=='bibliography_entry' else '{{sfn' if kind=='generated_note' else '<ref')
    if note_id is not None: assert doc.xpath('//li[@id=$x]',x=note_id),note_id
    data['definitions'].append({'id':f'd{len(data["definitions"])+1}','at':at,'in_window':data['article']['window'][0]<=cp<data['article']['window'][1],'kind':kind,'name':name,'group':'','templates':[{'at':loc(t,'{{')} for t in templates],'unstructured':unstructured,'render_note_id':note_id,'marker':'none','marker_text':None,'notes':notes})

def use(cp,target,ref_id,notes=''):
    if ref_id is not None: assert doc.xpath('//sup[@id=$x and contains(@typeof,"mw:Extension/ref")]',x=ref_id),ref_id
    data['uses'].append({'id':f'u{len(data["uses"])+1}','at':loc(cp),'binds_to':loc(target),'render_ref_id':ref_id,'notes':notes})

def identifier(cp,raw,key,scheme='doi',role='primary_citation_field',notes=''):
    data['identifiers'].append({'at':loc(cp,raw=raw),'scheme':scheme,'role':role,'key':key,'notes':notes})

def finish():
    data['definitions'].sort(key=lambda x:x['at']['cp']); data['uses'].sort(key=lambda x:x['at']['cp'])
    for field,prefix in [('definitions','d'),('uses','u')]:
        for n,item in enumerate(data[field],1): item['id']=prefix+str(n)
    def validate(x):
        if isinstance(x,dict):
            if 'cp' in x and 'quote' in x: assert text[x['cp']:x['cp']+len(x['quote'])]==x['quote']
            for v in x.values(): validate(v)
        elif isinstance(x,list):
            for v in x: validate(v)
    validate(data)
    for use_item in data['uses']:
        assert any(d['at']['cp']==use_item['binds_to']['cp'] for d in data['definitions'])
    data['provenance']['finished_at']=datetime.datetime.now(datetime.timezone.utc).isoformat()
    path=OUT/f'en-{data["article"]["revision_id"]}.json'
    path.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(path.name, 'definitions',len(data['definitions']),'uses',len(data['uses']),'identifiers',len(data['identifiers']))
