"""Serialize manually decided labels at explicitly inspected source positions.
No reference parsing or automatic classification is performed here.
"""
from pathlib import Path
import json
import hashlib
from datetime import datetime, timezone

ROOT = Path('/workspace/stage03-work/final-evaluators-initial')
OUT = ROOT / 'output-ar-2'
REVS = [76509905, 76545799, 76553307, 76927230, 76937308]
opened = []
for p in [ROOT/'ar-2-BRIEF.md', ROOT/'ANNOTATION_GUIDE.md', Path(__file__)]:
    opened.append({'path': str(p), 'sha256': hashlib.sha256(p.read_bytes()).hexdigest()})
sources, packets = {}, {}
for rev in REVS:
    folder = ROOT/'ar-2'/f'ar-{rev}'
    packet_path = folder/'packet.json'
    packet = json.loads(packet_path.read_text(encoding='utf-8'))
    packets[rev] = packet
    opened.append({'path': str(packet_path), 'sha256': hashlib.sha256(packet_path.read_bytes()).hexdigest()})
    for filename, key in [('source.wikitext', 'source_sha256'), ('render.html', 'html_sha256')]:
        p = folder/filename
        raw = p.read_bytes()
        assert hashlib.sha256(raw).hexdigest() == packet[key]
        opened.append({'path': str(p), 'sha256': packet[key]})
        if filename == 'source.wikitext':
            sources[rev] = raw.decode('utf-8')

for rev in REVS:
    s, p = sources[rev], packets[rev]
    start, end = p['window']['cp_start'], p['window']['cp_end']
    def loc(cp, quote=None):
        quote = s[cp:cp+55] if quote is None else quote
        assert s[cp:cp+len(quote)] == quote
        return {'cp': cp, 'quote': quote}
    definitions = []
    for i,(cp,tp,name,note) in enumerate(DEFS[rev],1):
        assert s.startswith('<ref',cp) and s.startswith('{{استشهاد',tp)
        definitions.append({'id':f'd{i}','at':loc(cp),'in_window':start <= cp < end,'kind':'note','name':name,'group':'','templates':[{'at':loc(tp)}],'unstructured':False,'render_note_id':'cite_note-'+note,'marker':'none','marker_text':None,'notes':''})
    uses = []
    for i,(cp,target,marker) in enumerate(USES[rev],1):
        assert start <= cp < end and s.startswith('<ref',cp)
        uses.append({'id':f'u{i}','at':loc(cp),'binds_to':loc(target),'render_ref_id':'cite_ref-'+marker,'notes':'Named use precedes its content-bearing definition.' if cp < target else ''})
    identifiers = []
    for cp,raw,role,notes in IDENTIFIERS[rev]:
        assert start <= cp < end
        identifiers.append({'at':loc(cp,raw),'scheme':'doi','role':role,'key':raw.lower(),'notes':notes})
    doc = {'format':'citerift-stage03-annotation-v1','article':{'wiki':'ar','revision_id':rev,'source_sha256':p['source_sha256'],'html_sha256':p['html_sha256'],'window':[start,end]},'provenance':{'agent':'stage03-evaluator-ar-2','started_at':'2026-10-31T06:01:04.335402+00:00','finished_at':datetime.now(timezone.utc).isoformat(),'files_opened':opened,'tools':'ChatGPT/Codex independent evaluator; functions.exec with exec_command, cat, isolated Python standard library and lxml for raw text searches and HTML element/attribute listings; apply_patch for this explicit manual-table serialization helper. No network, git, repository code, parser reuse, or engine output. started_at is the first captured UTC timestamp during initial packet discovery.','unresolved':'See unresolved_notes; publisher /article/ URL interpretation uncertainty for ar-76553307.' if rev==76553307 else 'No unresolved source bindings or warning classifications. Render-only infobox references are documented separately where applicable.'},'definitions':definitions,'uses':uses,'identifiers':identifiers,'unresolved_notes':UNRESOLVED[rev]}
    def validate_locations(x):
        if isinstance(x,dict):
            if 'cp' in x and 'quote' in x:
                assert s[x['cp']:x['cp']+len(x['quote'])] == x['quote']
            for value in x.values(): validate_locations(value)
        elif isinstance(x,list):
            for value in x: validate_locations(value)
    validate_locations(doc)
    path = OUT/f'ar-{rev}.json'
    path.write_text(json.dumps(doc,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(path.name,'definitions',len(definitions),'uses',len(uses),'identifiers',len(identifiers))
