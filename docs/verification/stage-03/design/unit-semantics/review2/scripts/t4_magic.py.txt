"""RV-4 rule beside magic words and parser functions: a hand-written citation (no link, no DOI) in a <ref>, with a
magic word in its text. With a Parsoid-like transclusion for the magic word (target with "function"), and without one."""
import json, tempfile
from pathlib import Path
from rv2lib import *

def fn_transclusion(n, wt, function, params, inner):
    part = {"target": {"wt": wt, "function": function}, "params": {k: {"wt": v} for k, v in params.items()}, "i": 0}
    dm = json.dumps({"parts": [{"template": part}]}, ensure_ascii=False).replace("'", "&#39;")
    return f'<span about="#mwt{n}" typeof="mw:Transclusion" data-mw=\'{dm}\'>{inner}</span>'
CIT = "Smith, J. (2001). ''History of"
cases = [
 ("{{PAGENAME}}", "PAGENAME", "pagename", {}, "SYNTHETIC"),
 ("{{!}}", "!", "!", {}, "|"),
 ("{{CURRENTYEAR}}", "CURRENTYEAR", "currentyear", {}, "2026"),
 ("{{#if:x|Volume 2}}", "#if:x", "if", {"1": "Volume 2"}, "Volume 2"),
 ("{{nbsp}}", "nbsp", None, {}, "&#160;"),
 ("{{#invoke:String|sub|abcdef|1|3}}", "#invoke:String", "invoke", {"1": "sub", "2": "abcdef", "3": "1", "4": "3"}, "abc"),
]
for src, wt, fn, params, out in cases:
    text = f"A<ref>{CIT} {src}''. Oxford: Clarendon Press.</ref>\n"
    for mode in ("wrapped", "not wrapped", "no render"):
        if mode == "no render":
            body = None
        elif mode == "wrapped":
            t = fn_transclusion(5, wt, fn, params, out) if fn else transclusion(5, wt, params, out)
            body = one_note("A", f"Smith, J. (2001). <i>History of {t}</i>. Oxford: Clarendon Press.")
        else:
            body = one_note("A", f"Smith, J. (2001). <i>History of {out}</i>. Oxford: Clarendon Press.")
        with tempfile.TemporaryDirectory() as d:
            doc = run(Path(d), text, body)
            s = summary(doc)
            print(f"{src:40s} {mode:12s} units={[u[1] for u in s['units']]} H={[h[1] for h in s['H']]}")
