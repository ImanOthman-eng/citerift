from citerift.wiki.canonical import digest

def redigest(d):
    d["extraction_digest"] = digest({k: v for k, v in d.items() if k != "extraction_digest"})
    return d

def set_state(d, state):
    m = d["markers"][0]
    old = m["state"]; m["state"] = state
    acc = d["accounting"]["markers"]; acc[old] -= 1
    if not acc[old]: del acc[old]
    acc[state] = acc.get(state, 0) + 1

# 1: hidden-only note; forge a shown warning by relabelling the hidden item (keeps the locator).
doc, rep, b = observe('Paper. <span style="display:none">Retracted</span>.')
print("original:", short(doc["markers"][0]), rep["verified"])
f = copy.deepcopy(doc)
ev = f["markers"][0]["evidence"][0]
ev.update(kind="warning", part="text"); ev.pop("hidden_basis", None)
set_state(f, "recognized_warning_text")
r = verify_bundle(b, redigest(f))
print("forge 1 (hidden -> shown, locator kept):", short(f["markers"][0]), "verified =", r["verified"])
