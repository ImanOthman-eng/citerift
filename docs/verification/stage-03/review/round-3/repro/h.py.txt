def tmpdir():

def note_body(inner, wrap=("", "")):
    return ("<p>A." + ref_sup(1, "cite_note-a-1", "cite_ref-a_1-0", name="a") + "</p>" + wrap[0]
            + references([note_li("cite_note-a-1", ["cite_ref-a_1-0"], inner)]) + wrap[1])

def observe(inner, wiki="en", wrap=("", ""), body=None, do_verify=True):
    b = bundle(tmpdir(), EN_TEXT if wiki == "en" else AR_TEXT, body or note_body(inner, wrap), wiki=wiki)
    doc = extract(b)
    rep = verify_bundle(b, doc) if do_verify else None
    return doc, rep, b

def short(m):
