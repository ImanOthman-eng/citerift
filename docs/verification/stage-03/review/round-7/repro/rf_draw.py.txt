def att(code=None, problem=None, body=100): return NS(to_json=lambda: {"code": code, "body_bytes": body, "retry_after_problem": problem}, )
def ans(code=None, nb=None, attempts=(), js=None):
    return NS(code=code, not_before=nb, attempts=list(attempts), body=b"{}" if code is None else None, json=js)
def mk(answers, t=T0):
    d = Path(tempfile.mkdtemp(dir=base)); c = Clock(t)
    it = iter(answers); calls = []
    def fetch(url): calls.append(url); return next(it)
    return sel.LiveGet(d, fetch, c, c.sleep), c, calls, d
