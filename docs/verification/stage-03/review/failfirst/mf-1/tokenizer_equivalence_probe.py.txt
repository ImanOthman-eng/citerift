import json
import sys
import time
from pathlib import Path

sys.path.insert(0, "docs/verification/stage-03/review/failfirst/doi-d1-d2")
texts = []
for p in sorted(Path("docs/verification/stage-03/evaluation").rglob("capture.json")):
    cap = json.loads(p.read_text())
    if not cap.get("source", {}).get("artifact"):
        continue
    t = (p.parent / "artifacts" / cap["source"]["artifact"]).read_bytes().decode()
    texts.append(t)
big = "".join(f'P{i}.<ref name="n{i}">{{{{cite journal |title=T{i} |doi=10.1234/x{i}}}}}</ref>\n' for i in range(20000))
texts.append(big)
from mwparserfromhell.parser._tokenizer import CTokenizer
from mwparserfromhell.parser.tokenizer import Tokenizer as PyTok

same = 0
tc = tp = 0
worst = 0
for t in texts:
    a = time.perf_counter()
    c = CTokenizer().tokenize(t, 0, False)
    tc += time.perf_counter() - a
    a = time.perf_counter()
    p = PyTok().tokenize(t, 0, False)
    d = time.perf_counter() - a
    tp += d
    worst = max(worst, d)
    same += [repr(x) for x in c] == [repr(x) for x in p]
print(
    len(texts), "texts; identical token streams:", same, "C total %.2fs py total %.2fs py worst %.2fs" % (tc, tp, worst)
)
