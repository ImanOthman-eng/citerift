def marker_for(note_text, wiki):
    text = "x<ref>{{cite journal|title=T}}</ref>" if wiki == "en" else "أ<ref>{{استشهاد بدورية|عنوان=ت}}</ref>"
    note = "cite_note-1"
    body = "<p>x" + ref_sup(1, note, "cite_ref-1") + "</p>" + references([note_li(note, ["cite_ref-1"], note_text)])
    d, _ = run(text, body, wiki)
    return [m["state"] for m in d["markers"]]
